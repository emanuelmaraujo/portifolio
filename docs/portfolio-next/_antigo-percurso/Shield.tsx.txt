"use client";

import { useEffect, useState, type CSSProperties } from "react";
import { PLANE, REST } from "./take";
import { FLEET, LAYERS, RINGS, THREATS } from "./defence";

type Copy = { label: string; sheet: string; done: string; sealed: string; reset: string; skip: string; on: string; off: string; auto: string; threat: string; turnOn: string; stopped: string; wrong: string; layers: string; closed: string; intro: string };

const css = (vars: Record<string, string | number>) => vars as CSSProperties;
const at = (angle: number, r: number) => { const a = (angle * Math.PI) / 180; return [Math.round((PLANE.cx + r * Math.cos(a)) * 10) / 10, Math.round((PLANE.cy - r * Math.sin(a)) * 10) / 10]; }; // rounded: server and browser must agree
/** One eighth of the ring round the model. */
const arc = (i: number) => {
  const mid = 247.5 - i * 45, [x0, y0] = at(mid - 19, RINGS.progress), [x1, y1] = at(mid + 19, RINGS.progress);
  return `M${x0.toFixed(1)} ${y0.toFixed(1)}A${RINGS.progress} ${RINGS.progress} 0 0 0 ${x1.toFixed(1)} ${y1.toFixed(1)}`;
};

/**
 * The second act. The sheet is a glass screen and an attack plays out on it: five threats arrive one after the other
 * and each is stopped by the layer made for it, which the visitor turns on. The other three layers come up on their
 * own. With all eight on, the dome closes over the model (Scroll plays it when the root says so).
 */
export default function Shield({ copy }: { copy: Copy }) {
  const [step, setStep] = useState(0);      // threats stopped so far
  const [picked, setPicked] = useState(""); // the layer being explained
  const [miss, setMiss] = useState(false);
  const total = THREATS.length, done = step >= total, threat = done ? null : THREATS[step];
  const isOn = (id: string) => step >= LAYERS.find(l => l.id === id)!.since;
  const count = LAYERS.filter(l => step >= l.since).length;
  const wanted = threat ? LAYERS.find(l => l.id === threat.layer)! : null;
  const shown = LAYERS.find(l => l.id === picked) ?? null;

  useEffect(() => {
    const root = document.querySelector<HTMLElement>(".px-p");
    if (root) root.dataset.shield = done ? "1" : "0";
  }, [done]);

  const press = (id: string) => {
    if (wanted && id === wanted.id) { setStep(step + 1); setPicked(id); setMiss(false); }
    else { setPicked(id); setMiss(!isOn(id) && !!wanted); }
  };
  const device = (i: number) => (i === FLEET.infected && step === 3 ? "bad" : i === FLEET.infected && step >= 4 ? "isolated" : i === FLEET.late && step === 2 ? "late" : step >= 3 ? "ok" : "idle");
  const [tx0, ty0] = threat ? at(threat.angle, threat.from) : [0, 0], [tx1, ty1] = threat ? at(threat.angle, threat.to) : [0, 0];
  const last = step > 0 ? THREATS[step - 1] : null;

  return (
    <>
      <div className="p-panel p-os" data-from={REST.protejo[0] - 0.012} data-to={REST.protejo[1]} data-done={done ? "1" : "0"} data-step={step}>
        <div className="p-os-plane" style={{ width: PLANE.w, height: PLANE.h }}>
          <svg className="p-os-lines" viewBox={`0 0 ${PLANE.w} ${PLANE.h}`} aria-hidden="true">
            <circle className="p-os-radar" cx={PLANE.cx} cy={PLANE.cy} r={RINGS.wall + 70} />
            <path className="p-os-sweep" d={`M${PLANE.cx} ${PLANE.cy}L${PLANE.cx + RINGS.wall + 70} ${PLANE.cy}`} />
            <circle className="p-os-wall" data-on={isOn("firewall") ? "1" : "0"} cx={PLANE.cx} cy={PLANE.cy} r={RINGS.wall} />
            <circle className="p-os-wall p-os-wall--inner" data-on={isOn("firewall") ? "1" : "0"} cx={PLANE.cx} cy={PLANE.cy} r={RINGS.wall - 14} />
            <circle className="p-os-door" data-on={isOn("identidade") ? "1" : "0"} cx={PLANE.cx} cy={PLANE.cy} r={RINGS.door} />
            <circle className="p-os-core" data-hit={step === 4 ? "1" : "0"} cx={PLANE.cx} cy={PLANE.cy} r={RINGS.progress - 26} />
            <circle className="p-os-restore" data-on={done ? "1" : "0"} cx={PLANE.cx} cy={PLANE.cy} r={RINGS.progress - 50} pathLength={100} />
            {LAYERS.map((l, i) => {
              const [x0, y0] = at(247.5 - i * 45, RINGS.progress + 8), [x1, y1] = at(l.angle, RINGS.badge - 58);
              return (
                <g key={l.id} data-on={step >= l.since ? "1" : "0"}>
                  <path className="p-os-seg" d={arc(i)} />
                  <line className="p-os-link" x1={x0} y1={y0} x2={x1} y2={y1} />
                </g>
              );
            })}
            {FLEET.angles.map((a, i) => {
              const [x, y] = at(a, FLEET.radius);
              return (
                <g className="p-os-device" key={a} data-state={device(i)} style={css({ "--i": i })}>
                  <rect x={x - 17} y={y - 12} width={34} height={24} rx={3} />
                  <line x1={x - 9} y1={y + 17} x2={x + 9} y2={y + 17} />
                  <circle cx={x} cy={y} r={30} />
                </g>
              );
            })}
            {threat && (
              <g className="p-os-threat" key={"threat" + step} style={css({ "--x0": `${tx0}px`, "--y0": `${ty0}px`, "--x1": `${tx1}px`, "--y1": `${ty1}px` })}>
                <line x1={tx0} y1={ty0} x2={tx1} y2={ty1} />
                <path d="M0 -15L15 0L0 15L-15 0Z" />
              </g>
            )}
            {last && (() => { const [x, y] = at(last.angle, last.to); return <circle className="p-os-burst" key={"burst" + step} cx={x} cy={y} r={20} />; })()}
          </svg>
          <p className="p-os-count" aria-hidden="true"><b>{count}</b>/{LAYERS.length}<span>{done ? copy.closed : copy.layers}</span></p>
          {LAYERS.map((l, i) => {
            const [x, y] = at(l.angle, RINGS.badge), on = step >= l.since;
            return (
              <button type="button" className={l.live ? "p-os-module" : "p-os-module p-os-module--auto"} key={l.id} style={css({ left: x + "px", top: y + "px", "--i": i })}
                aria-pressed={on} data-next={wanted?.id === l.id ? "1" : "0"} data-picked={picked === l.id ? "1" : "0"} onClick={() => press(l.id)}>
                <i aria-hidden="true">{"0" + (i + 1)}</i>
                <b>{l.name}</b>
                <span>{on ? copy.on : l.live ? copy.off : copy.auto}</span>
                <em aria-hidden="true">{l.role}</em>
              </button>
            );
          })}
        </div>
      </div>

      <section className="p-panel p-card p-card--right p-mission" data-from={REST.protejo[0] - 0.005} data-to={REST.protejo[1]} aria-labelledby="p-card-2">
        <p className="p-card-head"><span>{copy.label}</span><span>{copy.sheet}</span></p>
        <h2 id="p-card-2">{threat ? <><small>{copy.threat} {step + 1}/{total}</small>{threat.title}</> : copy.done}</h2>
        <p className="p-meter" role="img" aria-label={`${count} / ${LAYERS.length} ${copy.layers}`}>{LAYERS.map(l => <i data-on={step >= l.since ? "1" : "0"} key={l.id} />)}</p>
        <div className="p-brief" aria-live="polite">
          {shown && <p key={shown.id + step + (miss ? "m" : "")}>{miss && wanted ? <em>{copy.wrong} {wanted.name}. </em> : null}<b>{shown.name}</b>{shown.id === last?.layer && !miss ? <> · {copy.stopped}</> : null}<br />{shown.note}</p>}
          {done && !shown && <p key="done">{copy.sealed}</p>}
          {threat && <p key={"t" + step} className={shown ? "p-brief-next" : undefined}>{step === 0 && !shown ? copy.intro + " " : ""}{threat.text}</p>}
          {done && shown && <p key="sealed" className="p-brief-next">{copy.sealed}</p>}
        </div>
        <p className="p-links">
          {wanted && <button type="button" className="p-main" onClick={() => press(wanted.id)}>{copy.turnOn} {wanted.name}</button>}
          {!done && <button type="button" onClick={() => { setStep(total); setPicked(""); setMiss(false); }}>{copy.skip}</button>}
          {done && <button type="button" onClick={() => { setStep(0); setPicked(""); setMiss(false); }}>{copy.reset}</button>}
        </p>
      </section>
    </>
  );
}
